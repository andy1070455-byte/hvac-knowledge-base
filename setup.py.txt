note_content = """---
title: "單段蒸氣壓縮冷凍循環狀態點與效能分析"
date: 2026-11-16
category: "冷凍循環"
tags: ["熱力學", "冷凍空調", "P-h圖", "COP"]
summary: "探討理想與實際單段蒸氣壓縮冷凍循環之四大程序及其性能係數 (COP) 計算。"
---

> **核心邏輯概述**：冷凍循環利用冷媒相變化吸收與釋放熱量，透過壓縮機、冷凝器、膨脹閥與蒸發器達成連續移熱。

### 1. 核心原理與計算
* **程序 1→2（等熵壓縮）**：過熱蒸氣經壓縮機壓縮至高壓高溫狀態。
  $$\\dot{W}_{in} = \\dot{m} (h_2 - h_1)$$
* **程序 2→3（等壓冷凝）**：高壓冷媒於冷凝器中釋放熱量予環境（空氣或冷卻水）。
  $$\\dot{Q}_{out} = \\dot{m} (h_2 - h_3)$$
* **程序 3→4（等焓膨脹）**：高壓液體經膨脹閥降壓降溫，變為低壓濕蒸氣。
  $$h_3 \\approx h_4$$
* **程序 4→1（等壓蒸發）**：低壓濕蒸氣於蒸發器中自被冷卻空間吸熱。
  $$\\dot{Q}_{in} = \\dot{m} (h_1 - h_4)$$

![蒸氣壓縮冷凍循環 P-h 圖](../images/2026-11-16-vapor-compression-cycle-01.png)

### 2. 原始數據與熱流參數推導
* 性能係數（Coefficient of Performance, COP）：
  $$COP_{refrig} = \\frac{\\dot{Q}_{in}}{\\dot{W}_{in}} = \\frac{h_1 - h_4}{h_2 - h_1}$$
* 容許冷凍噸（réfrigération ton, RT）：
  $$1 \\text{ RT} = 3.517 \\text{ kW} = 12,000 \\text{ Btu/hr}$$

### 3. 補充說明與經驗參數除錯
* **過熱度與過冷度控制**：蒸發器出口適度保持 5~7°C 之過熱度，可避免壓縮機發生液擊（Liquid Slug）；冷凝器出口保持 3~5°C 之過冷度以提升系統冷房能力。
"""
with open("notes/2026-11-16-vapor-compression-cycle.md", "w", encoding="utf-8") as f:

# 7. 寫入 app.js
app_js = """document.addEventListener("DOMContentLoaded", () => {
    const cardGrid = document.getElementById("card-grid");
    const readerView = document.getElementById("reader-view");
    const markdownContent = document.getElementById("markdown-content");
    const searchInput = document.getElementById("search-input");
    const categoryFilters = document.getElementById("category-filters");
    const backBtn = document.getElementById("back-btn");
    const themeToggle = document.getElementById("theme-toggle");
    const filterBarContainer = document.querySelector(".filter-bar");

    let allNotes = [];
    let currentCategory = "all";

    const savedTheme = localStorage.getItem("theme") || "light";
    document.documentElement.setAttribute("data-theme", savedTheme);

    themeToggle.addEventListener("click", () => {
        const currentTheme = document.documentElement.getAttribute("data-theme");
        const newTheme = currentTheme === "dark" ? "light" : "dark";
        document.documentElement.setAttribute("data-theme", newTheme);
        localStorage.setItem("theme", newTheme);
    });

    fetch("data/notes.json")
        .then(response => response.json())
        .then(data => {
            allNotes = data;
            initCategories(data);
            renderCards(data);
        })
        .catch(err => console.error("無法載入筆記索引：", err));

    function initCategories(notes) {
        const categories = ["all", ...new Set(notes.map(n => n.category))];
        categoryFilters.innerHTML = categories.map(cat => `
            <button class="tag-btn ${cat === 'all' ? 'active' : ''}" data-category="${cat}">
                ${cat === 'all' ? '全部類別' : cat}
            </button>
        `).join("");

        categoryFilters.querySelectorAll(".tag-btn").forEach(btn => {
            btn.addEventListener("click", (e) => {
                categoryFilters.querySelectorAll(".tag-btn").forEach(b => b.classList.remove("active"));
                e.target.classList.add("active");
                currentCategory = e.target.getAttribute("data-category");
                filterAndRender();
            });
        });
    }

    function renderCards(notes) {
        if (notes.length === 0) {
            cardGrid.innerHTML = "<p>沒有找到相關的工程筆記。</p>";
            return;
        }

        cardGrid.innerHTML = notes.map(note => `
            <div class="note-card" data-filename="${note.filename}">
                <div class="note-meta">${note.date} · ${note.category}</div>
                <h3>${note.title}</h3>
                <p class="note-summary">${note.summary}</p>
            </div>
        `).join("");

        document.querySelectorAll(".note-card").forEach(card => {
            card.addEventListener("click", () => {
                const filename = card.getAttribute("data-filename");
                loadMarkdownNote(filename);
            });
        });
    }

    function loadMarkdownNote(filename) {
        fetch(`notes/${filename}`)
            .then(res => res.text())
            .then(mdText => {
                const cleanedMd = mdText.replace(/^---[\\s\\S]*?---/, "");
                markdownContent.innerHTML = marked.parse(cleanedMd);
                
                cardGrid.classList.add("hidden");
                filterBarContainer.classList.add("hidden");
                readerView.classList.remove("hidden");
                window.scrollTo({ top: 0, behavior: "smooth" });
            })
            .catch(err => alert("載入筆記檔案失敗！"));
    }

    backBtn.addEventListener("click", () => {
        readerView.classList.add("hidden");
        cardGrid.classList.remove("hidden");
        filterBarContainer.classList.remove("hidden");
    });

    function filterAndRender() {
        const keyword = searchInput.value.toLowerCase();
        const filtered = allNotes.filter(note => {
            const matchCategory = currentCategory === "all" || note.category === currentCategory;
            const matchKeyword = note.title.toLowerCase().includes(keyword) || 
                                 note.summary.toLowerCase().includes(keyword) ||
                                 note.tags.some(t => t.toLowerCase().includes(keyword));
            return matchCategory && matchKeyword;
        });
        renderCards(filtered);
    }

    searchInput.addEventListener("input", filterAndRender);
});
"""
with open("app.js", "w", encoding="utf-8") as f:
    f.write(app_js.strip())
